239 def __init__(
240 self,
241 graph_id: int,
242 edge_type_id: int,
243 rank: int,
244 src_id: int,
245 dst_id: int,
246 properties: Dict[str, "ValueWrapper"],
247 graph_schemas: ResultGraphSchemas,
248 ):
249 """Edge is a wrapper around the Edge type returned by nebula-graph"""
250 super().__init__()
251 self.graph_id = graph_id
252 graph_schema = graph_schemas.get_graph_schema(graph_id)
253 self.graph_name = graph_schema.get_graph_name()
254 self.edge_type_id = edge_type_id
255 no_directed_type_id = edge_type_id & 0x3FFFFFFF
256 edge_schema = graph_schema.get_edge_schema(no_directed_type_id)
257 self.edge_type_name = edge_schema.get_edge_type_name()
258 self.labels = edge_schema.get_edge_labels()
259 self.rank = rank
260 self.properties = properties
261
262 edge_type_move_bits = EDGE_TYPE_ID_SIZE * 8 - 2
263 direction_bits = (edge_type_id >> edge_type_move_bits) & 0x3
264
265 if direction_bits == 0x0:
266 self.direction = "OUTGOING"
267 elif direction_bits == 0x1:
268 self.direction = "INCOMING"
269 elif direction_bits in (0x2, 0x3):
270 self.direction = "UNDIRECTED"
271 else:
272 self.direction = "KNOWN"
273
274 if self.direction == "INCOMING":
275 self.src_id = dst_id
276 self.dst_id = src_id
277 else:
278 self.src_id = src_id
279 self.dst_id = dst_id

462class Path(CompositeDataObject):
463 def __init__(self, values: list["ValueWrapper"]):
464 self.values = values
465 self.nodes: list[Node] = []
466 self.edges: list[Edge] = []
467
468 for value in values:
469 if value.is_node():
470 self.nodes.append(value.as_node())
471 else:
472 self.edges.append(value.as_edge())
473
474 def get_nodes(self) -> list[Node]:
475 """Create a list over the nodes in this path, nodes will appear in the same order as they appear
476 in the path.
477
478 Returns
479 -------
480 List[Node]: a List of all nodes in this path
481
482 """
483 return self.nodes
484
485 def get_edges(self) -> list[Edge]:
486 """Create a list over the edges in this path. The edges will appear
487 in the same order as they appear in the path.
488
489 Returns
490 -------
491 List[Edge]: a List of all edges in this path
492
493 """
494 return self.edges
495
496 def get_values(self) -> list["ValueWrapper"]:
497 """Create a list over the nodes and edges in this path. The value will appear
498 in the same order as they appear in the path. The first value will be Node type, then the
499 next one will be Edge type, and next one will be Node type.
500
501 Returns
502 -------
503 List[ValueWrapper]: a List of all values in this path
504
505 """
506 return self.values
507
508 def cast_primitive(
509 self,
510 fields: Optional[Set[PathPrimitiveField]] = None,
511 ) -> PathPrimitive:
512 """Convert path to a dictionary format with nodes and edges.
513
514 Args:
515 ----
516 fields: List of fields to include in the dictionary.
517
518 Example path string representation:
519 (289226172909223937@Movie:Movie{id:91,name:Unpromised Land})-
520 [0@WithGenre:WithGenre{}]->
521 (289483299716333572@Genre:Genre{id:101,name:Staged Documentary})
522
523 Returns:
524 -------
525 Dict containing the requested path information. Only fields specified in the
526 fields parameter will be included.
527
528 """
529 fields = fields or set(PathPrimitiveField)
530 mapping = {
531 PathPrimitiveField.NODES: lambda: [
532 node.cast_primitive() for node in self.nodes
533 ],
534 PathPrimitiveField.EDGES: lambda: [
535 edge.cast_primitive() for edge in self.edges
536 ],
537 PathPrimitiveField.LENGTH: lambda: len(self.nodes),
538 PathPrimitiveField.START_NODE: lambda: self.nodes[0].cast_primitive(),
539 PathPrimitiveField.END_NODE: lambda: self.nodes[-1].cast_primitive(),
540 PathPrimitiveField.STRING_REPRESENTATION: lambda: str(self),
541 }
542 result: PathPrimitive = {}
543 for field in fields:
544 result[field.value] = mapping[field]()
545
546 return result
547
548 def __eq__(self, other):
549 if self is other:
550 return True
551 if other is None or not isinstance(other, Path):
552 return False
553 return self.values == other.values
554
555 def __hash__(self):
556 return hash(self.values)
557
558 def __str__(self):
559 if not self.values:
560 return "()"
561
562 prefix_node = self.nodes[0]
563 prefix_node_prop_strs = []
564 prefix_node_props = prefix_node.get_properties()
565 for key in prefix_node_props:
566 prefix_node_prop_strs.append(f"{key}:{prefix_node_props[key].cast()}")
567
568 # Just one node in the path
569 if not self.edges:
570 template = "(%d@%s:%s{%s})"
571 return template % (
572 prefix_node.get_id(),
573 prefix_node.get_type(),
574 prefix_node.get_type(),
575 ",".join(prefix_node_prop_strs),
576 )
577
578 edge_strs = []
579 for i, edge in enumerate(self.edges):
580 edge_prop_strs = []
581 props = edge.get_properties()
582 for key in props:
583 edge_prop_strs.append(f"{key}:{props[key].cast()}")
584
585 suffix_node = self.nodes[i + 1]
586 suffix_node_prop_strs = []
587 suffix_node_props = suffix_node.get_properties()
588 for key in suffix_node_props:
589 suffix_node_prop_strs.append(f"{key}:{suffix_node_props[key].cast()}")
590
591 if i == 0:
592 template = "(%d@%s:%s{%s})~[%d@%s:%s{%s}]~(%d@%s:%s{%s})"
593 if edge.is_directed():
594 if edge.get_src_id() == prefix_node.get_id():
595 template = "(%d@%s:%s{%s})-[%d@%s:%s{%s}]->(%d@%s:%s{%s})"
596 else:
597 template = "(%d@%s:%s{%s})<-[%d@%s:%s{%s}]-(%d@%s:%s{%s})"
598
599 edge_strs.append(
600 template
601 % (
602 prefix_node.get_id(),
603 prefix_node.get_type(),
604 prefix_node.get_type(),
605 ",".join(prefix_node_prop_strs),
606 edge.get_rank(),
607 edge.get_type(),
608 edge.get_type(),
609 ",".join(edge_prop_strs),
610 suffix_node.get_id(),
611 suffix_node.get_type(),
612 suffix_node.get_type(),
613 ",".join(suffix_node_prop_strs),
614 ),
615 )
616 else:
617 template = "~[%d@%s:%s{%s}]~(%d@%s:%s{%s})"
618 if edge.is_directed():
619 if edge.get_dst_id() == suffix_node.get_id():
620 template = "-[%d@%s:%s{%s}]->(%d@%s:%s{%s})"
621 else:
622 template = "<-[%d@%s:%s{%s}]-(%d@%s:%s{%s})"
623 edge_strs.append(
624 template
625 % (
626 edge.get_rank(),
627 edge.get_type(),
628 edge.get_type(),
629 ",".join(edge_prop_strs),
630 suffix_node.get_id(),
631 suffix_node.get_type(),
632 suffix_node.get_type(),
633 ",".join(suffix_node_prop_strs),
634 ),
635 )
636
637 return "".join(edge_strs)

640class NRecord(CompositeDataObject):
641 def __init__(self, properties: Dict[str, "ValueWrapper"]):
642 self.type = ColumnType.RECORD
643 self.map: Dict[str, "ValueWrapper"] = {}
644 if properties:
645 self.map.update(properties)
646
647 def contains_key(self, key: str) -> bool:
648 """Returns true if this record contains a mapping for the specified key. The key cannot be null.
649
650 Args:
651 ----
652 key: key whose presence in this record is to be checked
653 Returns:
654 bool: true if this record contains a mapping for the specified key
655 Raises:
656 NullPointerException: if the specified key is null
657
658 """
659 if key is None:
660 raise ValueError("null map key")
661 return key in self.map
662
663 def get_value(self, key: str) -> Optional["ValueWrapper"]:
664 """Get the Value of specified key in this record
665
666 Args:
667 ----
668 key: key whose corresponding value in this record will be returned
669 Returns:
670 The ValueWrapper of the specified key
671
672 """
673 if not self.contains_key(key):
674 return None
675 return self.map[key]
676
677 def is_empty(self) -> bool:
678 """Returns true if this record has no values
679
680 Returns
681 -------
682 bool: true if this record contains no key-value mappings
683
684 """
685 return not self.map
686
687 def size(self) -> int:
688 """Get the number of key-value mappings in this record.
689
690 Returns
691 -------
692 int: the number of key-value mappings in this record
693
694 """
695 return len(self.map)
696
697 def get_values_map(self) -> Dict[str, "ValueWrapper"]:
698 """Get the Map object for this record
699
700 Returns
701 -------
702 Dict: Map for this record
703
704 """
705 return self.map
706
707 def cast_primitive(self) -> Dict[str, Any]:
708 """Convert record(Map or Dict actually) to a dictionary format with property name -> primitive value
709
710 Returns
711 -------
712 Dict: A dictionary containing property name -> primitive value
713
714 """
715 return {key: value.cast_primitive() for key, value in self.map.items()}
716
717 def __str__(self):
718 return str(self.cast_primitivecast_primitive())
719
720 def __eq__(self, other):
721 if self is other:
722 return True
723 if not isinstance(other, NRecord):
724 return False
725 return self.map == other.get_values_map()
726

728 return hash(frozenset(self.map.items()))

731class NDuration(BaseDataObject):
732 def __init__(self, is_month_based: bool, year: int, month: int, day: int,
733 hour: int, minute: int, seconds: int, microseconds: int):
734 self.is_month_based = is_month_based
735 self.year = year
736 self.month = month
737 self.day = day
738 self.hour = hour
739 self.minute = minute
740 self.second = seconds
741 self.microsec = microseconds
742
743 def get_year(self) -> int:
744 return self.year

746 def get_month(self) -> int:
747 return self.month
748
749 def get_day(self) -> int:
750 return self.day
751
752 def get_hour(self) -> int:
753 return self.hour
754
755 def get_minute(self) -> int:
756 return self.minute
757
758 def get_second(self) -> int:
759 return self.second
760
761 def get_microsecond(self) -> int:
762 return self.microsec
763
764 def __str__(self) -> str:
765 duration_str = ["P"]
766 if self.is_month_based:
767 if self.year != 0:
768 duration_str.append(f"{self.year}Y")
769 if self.month != 0 or self.year == 0:
770 duration_str.append(f"{self.month}M")
771 else:
772 if self.day != 0:
773 duration_str.append(f"{self.day}D")
774
775 # Only add T if we have time components or zero duration
776 if (
777 self.day == 0
778 or self.hour != 0
779 or self.minute != 0
780 or self.second != 0
781 or self.microsec != 0
782 ):
783 duration_str.append("T")
784
785 if self.hour != 0:
786 duration_str.append(f"{self.hour}H")
787 if self.minute != 0:
788 duration_str.append(f"{self.minute}M")
789
790 if (
791 self.second != 0
792 or self.microsec != 0
793 or (
794 self.day == 0
795 and self.hour == 0
796 and self.minute == 0
797 and self.second == 0
798 )
799 ):
800 if self.microsec == 0:
801 duration_str.append(f"{self.second}S")
802 else:
803 total_microseconds = self.second * 1000000 + self.microsec
804 is_minus = total_microseconds < 0
805 if is_minus:
806 total_microseconds = -total_microseconds
807 seconds = total_microseconds // 1000000
808 microsec = total_microseconds % 1000000
809
810 if is_minus:
811 num_str = f"-{seconds}.{microsec:06d}"
812 else:
813 num_str = f"{seconds}.{microsec:06d}"
814
815 # Remove trailing zeros
816 while num_str[-1] == "0":
817 num_str = num_str[:-1]
818 duration_str.append(f"{num_str}S")
819
820 return "".join(duration_str)
821
822 def __eq__(self, other):
823 if not isinstance(other, NDuration):
824 return False
825 return (
826 self.is_month_based == other.is_month_based
827 and self.year == other.get_year()
828 and self.month == other.get_month()
829 and self.day == other.get_day()
830 and self.hour == other.get_hour()
831 and self.minute == other.get_minute()
832 and self.second == other.get_second()
833 and self.microsec == other.get_microsecond()
834 )
835
836 def __hash__(self):
837 return hash(
838 (
839 self.is_month_based,
840 self.year,
841 self.month,
842 self.day,
843 self.hour,
844 self.minute,
845 self.second,
846 self.microsec,
847 ),
848 )
